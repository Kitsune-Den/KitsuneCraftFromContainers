# Kitsune Craft From Containers
#
# On a server, the server's copy of this file decides: clients use the
# smaller range and only the sources both sides turn on.

# How far (in blocks) a container can be from you and still feed crafting (1-64).
range=30

# Output slots of forges, campfires, workbenches, chem stations and cement
# mixers. Never their input, fuel or tool slots.
workstations=true

# Output slots of apiaries, chicken coops and dew collectors.
collectors=true

# World containers (a POI's cabinets, fridges, desks...) inside a land claim
# you own or are allied to. Player-placed storage always counts.
claimed=true

# Vehicle and drone storage that isn't locked against you.
vehicles=true

# Repairing and upgrading blocks, and repairing tools, weapons, armor,
# vehicles and drones, use the same containers as crafting.
repair=true

# Reloading guns, bows and launchers. Also counts toward the ammo shown on
# the HUD and in the ammo radial.
reload=true

# Refueling vehicles from the radial menu.
refuel=true

# Paying traders and other players' vending machines.
trader=true

# Storage, vehicles and drones locked against you count anyway when their
# owner is in your party.
party=false

# Storage, workstations and collectors count only inside a land claim you own
# or are allied to, even ones you placed. Vehicles and drones aren't affected.
claimonly=false

# Repair on the land claim block (and kcfc repair) fixes every damaged block
# on the claim, paying from the containers on it. Server side.
claimrepair=true

# Allow claim repair during a blood moon.
claimrepairbloodmoon=false
